from pathlib import Path

path = Path("requirements.txt")
text = path.read_text(encoding="utf-8")

if not any(
    line.strip().lower().startswith(("streamlit==", "streamlit>=", "streamlit~="))
    for line in text.splitlines()
):
    import streamlit
    path.write_text(
        text.rstrip() + f"\nstreamlit=={streamlit.__version__}\n",
        encoding="utf-8",
    )
    print("Added Streamlit to requirements.txt")
else:
    print("Streamlit already listed")
PY
